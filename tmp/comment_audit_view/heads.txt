/**
 * @file    forward_solution.h
 * @brief   平面五连杆机构正运动学解算
 * @Author  SYi777
 * @Email   jyh1621839418@163.com
 * @Date    2026-6-17
 *
 * 机构拓扑
 *
 *             O (固定坐标系原点, 基座)
 *            / \
 *          l1   l2      ← 输入杆 (主动)
 *          /     \
 *         A       C     ← 主动关节 (A = 左驱动, C = 右驱动)
 *          \     /
 *          l3   l4      ← 从动杆 (被动)
 *            \ /
 *             B         ← 末端执行点
 *
 * ─── 角度定义 (所有角度以逆时针为正, 参考轴为 +X) ───
 *
 *   alpha       : 输入杆 OA 的转角, 从 +X 轴逆时针量至 OA
 *   beta        : 输入杆 OC 的转角, 从 +X 轴逆时针量至 OC
 *   th1         : 从动杆 AB 的全局绝对角, 从 +X 轴逆时针量至 AB, 范围 [0, 2*pi)
 *   th2         : 从动杆 CB 的全局绝对角, 从 +X 轴逆时针量至 CB, 范围 [0, 2*pi)
 *   delta_input : 主动杆(曲柄 OA 与 OC)相对夹角, [0, pi]. 0=两曲柄同向(伸腿极限),
 *                 越大曲柄越张开、腿越短; pi=两曲柄反向共线(收腿翻转死点). 用于主动侧限位.
 *   delta_passive: 从动杆(AB 与 CB)相对夹角, [0, pi]. 0=两杆平行(伸腿极限),
 *                 越大腿越短; pi=两杆共线反向(雅可比奇异, sin(th2-th1)=0). 用于从动侧干涉判断.
 *
 *   坐标系: 从车辆左侧看，车头在画面左边；O=(0,0)，+X 朝车头（画面左），+Y 朝下
 */
==================== Kinematics\inverse_solution.h
/**
 * @file    inverse_solution.h
 * @brief   平面五连杆机构逆运动学解算
 * @Author  SYi777
 * @Email   jyh1621839418@163.com
 * @Date    2026-6-17
 *
 * 逆解公式 (L1～L4 与坐标均使用 mm，角度使用 rad)
 *
 *   已知目标轮心 B=(xb, yb):
 *     Rob = sqrt(xb*xb + yb*yb)
 *     psi = atan2(yb, xb)
 *
 *   用三角 OAB、OCB 的余弦定理:
 *     phi1 = acos((L1*L1 + Rob*Rob - L3*L3)/(2*L1*Rob))
 *     phi2 = acos((L2*L2 + Rob*Rob - L4*L4)/(2*L2*Rob))
 *
 *   两个输入角各有两个候选:
 *     alpha = wrap_pos(psi + phi1) 或 wrap_pos(psi - phi1)
 *     beta  = wrap_pos(psi + phi2) 或 wrap_pos(psi - phi2)
 *   因此最多有四组 (alpha, beta) 候选。
 *
 *   候选筛选顺序与 fivebar_ik 实现一致:
 *   1) da=wrap(alpha-psi), db=wrap(beta-psi)。若 da、db 同号则淘汰。
 *   2) cross_oa_oc=A.x*C.y-A.y*C.x=L1*L2*sin(beta-alpha)。
 *      按 ASSEMBLY_MODE 所需的叉积符号选择装配解。
 *   3) 用 fivebar_fk(alpha,beta) 验证末端坐标:
 *      dx=fk.B.x-xb, dy=fk.B.y-yb;
 *      dx*dx+dy*dy <= IK_POS_TOL*IK_POS_TOL。
 *
 *   坐标系: 从车辆左侧看，车头在画面左边；O=(0,0)，+X 朝车头（画面左），+Y 朝下，角度逆时针为正
 */
==================== Kinematics\forward_solution.c
/**
 * @file    forward_solution.c
 * @brief   平面五连杆机构正运动学解算
 * @Author  SYi777
 * @Email   jyh1621839418@163.com
 * @Date    2026-6-17
 *
 * 机构拓扑
 *
 *             O (固定坐标系原点, 基座)
 *            / \
 *          l1   l2      ← 输入杆 (主动)
 *          /     \
 *         A       C     ← 主动关节 (A = 左驱动, C = 右驱动)
 *          \     /
 *          l3   l4      ← 从动杆 (被动)
 *            \ /
 *             B         ← 末端执行点
 *
 * ─── 角度定义 (所有角度以逆时针为正, 参考轴为 +X) ───
 *
 *   alpha       : 输入杆 OA 的转角, 从 +X 轴逆时针量至 OA
 *   beta        : 输入杆 OC 的转角, 从 +X 轴逆时针量至 OC
 *   th1         : 从动杆 AB 的全局绝对角, 从 +X 轴逆时针量至 AB, 范围 [0, 2*pi)
 *   th2         : 从动杆 CB 的全局绝对角, 从 +X 轴逆时针量至 CB, 范围 [0, 2*pi)
 *   delta_input : 主动杆(曲柄 OA 与 OC)相对夹角, [0, pi]. 0=两曲柄同向(伸腿极限),
 *                 越大曲柄越张开、腿越短; pi=两曲柄反向共线(收腿翻转死点). 用于主动侧限位.
 *   delta_passive: 从动杆(AB 与 CB)相对夹角, [0, pi]. 0=两杆平行(伸腿极限),
 *                 越大腿越短; pi=两杆共线反向(雅可比奇异, sin(th2-th1)=0). 用于从动侧干涉判断.
 *
 *   坐标系: 从车辆左侧看，车头在画面左边；O=(0,0)，+X 朝车头（画面左），+Y 朝下
 *
 * ─── 重要说明 ───
 *   1. th1、th2 是全局绝对角，可落在 [0, 2*pi) 任意象限，大于 pi 属于正常现象
 *   2. 严禁对绝对角直接减 π，会导致杆朝向反向、末端坐标完全错误
 *   3. 干涉/限位判断必须使用「相对夹角」，与连杆整体姿态无关
 *   4. 解选择使用「叉积判据」(B 相对 AC 连线的位置)，与绝对角大小无关
 *   5. 叉积判据天然免疫连杆整体翻转 180° 的影响
 */
