/**
*@file 3508_CtrlApplications.c
*@brief C620轮电机力矩下发与发送状态记录
*@Author SYi777
*@Email jyh1621839418@163.com
*@Date
*
*左右轮共用 CAN ID 0x200 控制帧，反馈 ID 分别为 0x201、0x202。
*右轮电流方向在 WheelTorque_Send 中取反。
**/
#include "3508_CtrlApplications.h"

volatile wheel_drive_diag_t wheel_drive_diag;

motor_measure_t motor_measure_L;   // 左轮
motor_measure_t motor_measure_R;   // 右轮

/**
 * @brief  下发左右轮力矩对应的 C620 电流帧
 * @param  hfdcan  轮电机 CAN 句柄
 * @param  T_left  左轮力矩指令 [N*m]
 * @param  T_right  右轮力矩指令 [N*m]
 * @return 无
 *
 * 右轮电流在下发前取反；发送结果记录到 wheel_drive_diag。
 */
void WheelTorque_Send(FDCAN_HandleTypeDef *hfdcan, float T_left, float T_right)
{
    int16_t cur[4];
    cur[0] = (int16_t)(T_left*3000);   // 左轮
    cur[1] = (int16_t)(-T_right*3000);   // 右轮
    cur[2] = 0;
    cur[3] = 0;

    FDCAN_TxHeaderTypeDef tx_header;
    uint8_t data[8];
    for (int i = 0; i < 4; i++)          // C620 0x200 帧按高字节在前打包
    {
        uint16_t raw = (uint16_t)cur[i];
        data[2*i]     = (uint8_t)(raw >> 8);
        data[2*i + 1] = (uint8_t)(raw & 0xFF);
    }
    tx_header.Identifier          = CTRL_ID;                 // 0x200
    tx_header.IdType              = FDCAN_STANDARD_ID;
    tx_header.TxFrameType         = FDCAN_DATA_FRAME;
    tx_header.DataLength          = FDCAN_DLC_BYTES_8;
    tx_header.ErrorStateIndicator = FDCAN_ESI_ACTIVE;
    tx_header.BitRateSwitch       = FDCAN_BRS_OFF;
    tx_header.FDFormat            = FDCAN_CLASSIC_CAN;
    tx_header.TxEventFifoControl  = FDCAN_NO_TX_EVENTS;
    tx_header.MessageMarker       = 0;
    wheel_drive_diag.current_cmd_raw[0] = cur[0];
    wheel_drive_diag.current_cmd_raw[1] = cur[1];
    HAL_StatusTypeDef status = HAL_FDCAN_AddMessageToTxFifoQ(hfdcan, &tx_header, data);
    wheel_drive_diag.tx_status = (uint32_t)status;
    if (status == HAL_OK)
    {
        wheel_drive_diag.tx_cnt++;
        wheel_drive_diag.last_cmd_tick = HAL_GetTick();
    }
    else wheel_drive_diag.tx_fail_cnt++;
}

/**
*@file Remote_control.c
*@brief 遥控器 SBUS 数据接收与解析
*@Author 王草凡 SYi777
*@Email jyh1621839418@163.com
*@Date 2026-10-13
*
*参考王草凡的 C 板实现，移植到 H7 并使用 HAL 初始化 DMA，添加遥控器离线保护
**/
// 参考王草凡的遥控器协议说明
// 移植 C 板遥控器解析逻辑，使用 HAL 初始化 H7 的 DMA
#include "Remote_control.h"
#include "struct_typedef.h"

volatile uint8_t rc_online = 0;			// 遥控器在线标志，0 离线 1 在线
volatile uint8_t flag      = 0;			// 中断内部标志位
volatile uint32_t error_count = 0;		// 未进中断计数

RC_ctrl_t remote_ctrl;// 遥控器控制变量
__attribute__ ((section(".ARM.__at_0x24000000"))) uint8_t SBUS_MultiRx_Buf[2][SBUS_RX_BUF_NUM];// 双缓冲接收 SBUS 原始数据，防止DMA传输越界
// 内部函数声明
static void SBUS_TO_RC(volatile const uint8_t *sbus_buf, RC_ctrl_t *rc_ctrl);
static int16_t RC_abs(int16_t value);
static void USER_USART5_RxHandler(UART_HandleTypeDef *huart, uint16_t Size);
/**
 * @brief  解析一帧 SBUS 数据并更新遥控器状态
 * @param  sbus_buf  收到的 SBUS 原始数据
 * @param  remote_ctrl  输出的遥控器状态
 * @return 无
 */
static void SBUS_TO_RC(volatile const uint8_t *sbus_buf, RC_ctrl_t *remote_ctrl)
{
    if (sbus_buf == NULL || remote_ctrl == NULL)
    {
        return;
    }

    remote_ctrl->rc.ch[0] = (sbus_buf[0] | (sbus_buf[1] << 8)) & 0x07ff;        // 遥控器通道 0
    remote_ctrl->rc.ch[1] = ((sbus_buf[1] >> 3) | (sbus_buf[2] << 5)) & 0x07ff; // 遥控器通道 1
    remote_ctrl->rc.ch[2] = ((sbus_buf[2] >> 6) | (sbus_buf[3] << 2) |          // 遥控器通道 2
                         (sbus_buf[4] << 10)) &0x07ff;
    remote_ctrl->rc.ch[3] = ((sbus_buf[4] >> 1) | (sbus_buf[5] << 7)) & 0x07ff; // 遥控器通道 3
    remote_ctrl->rc.s[0] = ((sbus_buf[5] >> 4) & 0x0003);                  		// 左拨杆
    remote_ctrl->rc.s[1] = ((sbus_buf[5] >> 4) & 0x000C) >> 2;                  // 右拨杆
    remote_ctrl->mouse.x = sbus_buf[6] | (sbus_buf[7] << 8);                    // 鼠标 X 轴
    remote_ctrl->mouse.y = sbus_buf[8] | (sbus_buf[9] << 8);                    // 鼠标 Y 轴
    remote_ctrl->mouse.z = sbus_buf[10] | (sbus_buf[11] << 8);                  // 鼠标 Z 轴
    remote_ctrl->mouse.press_l = sbus_buf[12];                                  // 鼠标左键
    remote_ctrl->mouse.press_r = sbus_buf[13];                                  // 鼠标右键
    remote_ctrl->key.v = sbus_buf[14] | (sbus_buf[15] << 8);                    // 键盘按键
    remote_ctrl->rc.ch[4] = sbus_buf[16] | (sbus_buf[17] << 8);                 // 遥控器通道 4

    remote_ctrl->rc.ch[0] -= RC_CH_VALUE_OFFSET;
    remote_ctrl->rc.ch[1] -= RC_CH_VALUE_OFFSET;
    remote_ctrl->rc.ch[2] -= RC_CH_VALUE_OFFSET;
    remote_ctrl->rc.ch[3] -= RC_CH_VALUE_OFFSET;
    remote_ctrl->rc.ch[4] -= RC_CH_VALUE_OFFSET;
}

// 计算遥控器通道值的绝对值
static int16_t RC_abs(int16_t value)
{
    return (value > 0) ? value : -value;
}

/**
 * @brief  处理 UART5 DMA 双缓冲接收事件
 * @param  huart  UART 句柄
 * @param  Size  本次接收的字节数
 * @return 无
 *
 * 根据 DMA 目标缓冲区位选择已完成接收的缓冲区。
 */
static void USER_USART5_RxHandler(UART_HandleTypeDef *huart, uint16_t Size)
{
    flag = 1;   // 收到接收事件 → 遥控器在线喂狗
	
	
    if (((((DMA_Stream_TypeDef *)huart->hdmarx->Instance)->CR) & DMA_SxCR_CT) == RESET)
    {
        // 缓冲区0 接收完成
        __HAL_DMA_DISABLE(huart->hdmarx);
        ((DMA_Stream_TypeDef *)huart->hdmarx->Instance)->CR |= DMA_SxCR_CT;
        __HAL_DMA_SET_COUNTER(huart->hdmarx, SBUS_RX_BUF_NUM);
        if (Size == RC_FRAME_LENGTH)
        {
            SBUS_TO_RC(SBUS_MultiRx_Buf[0], &remote_ctrl);
        }
    }
    else
    {
        // 缓冲区1 接收完成
        __HAL_DMA_DISABLE(huart->hdmarx);
        ((DMA_Stream_TypeDef *)huart->hdmarx->Instance)->CR &= ~(DMA_SxCR_CT);
        __HAL_DMA_SET_COUNTER(huart->hdmarx, SBUS_RX_BUF_NUM);
        if (Size == RC_FRAME_LENGTH)
        {
            SBUS_TO_RC(SBUS_MultiRx_Buf[1], &remote_ctrl);
        }
    }
    __HAL_DMA_ENABLE(huart->hdmarx);
}

/**
 * @brief  将 UART5 接收事件转交给遥控器处理函数
 * @param  huart  触发事件的 UART 句柄
 * @param  Size  本次接收的字节数
 * @return 无
 */
void HAL_UARTEx_RxEventCallback(UART_HandleTypeDef *huart, uint16_t Size)
{
    if (huart == &huart5)
    {
        USER_USART5_RxHandler(huart, Size);
    }
}
// 获取全局遥控器状态的只读指针
const RC_ctrl_t *get_remote_control_point(void)
{
    return &remote_ctrl;
}

/**
 * @brief  检测遥控器通道与拨杆数据是否越界
 * @return 0 表示正常，1 表示错误且遥控器状态已清零
 */
uint8_t RC_data_is_error(void)
{
    if (RC_abs(remote_ctrl.rc.ch[0]) > RC_CHANNAL_ERROR_VALUE) goto error;
    if (RC_abs(remote_ctrl.rc.ch[1]) > RC_CHANNAL_ERROR_VALUE) goto error;
    if (RC_abs(remote_ctrl.rc.ch[2]) > RC_CHANNAL_ERROR_VALUE) goto error;
    if (RC_abs(remote_ctrl.rc.ch[3]) > RC_CHANNAL_ERROR_VALUE) goto error;
    if (remote_ctrl.rc.s[0] == 0) goto error;
    if (remote_ctrl.rc.s[1] == 0) goto error;
    return 0;
error:
    remote_ctrl.rc.ch[0] = 0;
    remote_ctrl.rc.ch[1] = 0;
    remote_ctrl.rc.ch[2] = 0;
    remote_ctrl.rc.ch[3] = 0;
    remote_ctrl.rc.ch[4] = 0;
    remote_ctrl.rc.s[0] = RC_SW_DOWN;
    remote_ctrl.rc.s[1] = RC_SW_DOWN;
    remote_ctrl.mouse.x = 0;
    remote_ctrl.mouse.y = 0;
    remote_ctrl.mouse.z = 0;
    remote_ctrl.mouse.press_l = 0;
    remote_ctrl.mouse.press_r = 0;
    remote_ctrl.key.v = 0;
    return 1;
}
// 重启遥控器 DMA 双缓冲接收
void slove_data_error(void)
{
    RC_restart(SBUS_RX_BUF_NUM, SBUS_MultiRx_Buf[0], SBUS_MultiRx_Buf[1]);
}
// 初始化 UART5 DMA 双缓冲接收
void RemoteContrlInit(void)
{
	// 启动 DMA 双缓冲接收
	USART_RxDMA_DoubleBuffer_Init(&huart5,(uint32_t *)SBUS_MultiRx_Buf[0],(uint32_t *)SBUS_MultiRx_Buf[1],SBUS_RX_BUF_NUM);
}

volatile uint8_t* GetRCOnlineFlag(void)
{
	return &rc_online;
}

/**
 * @brief  遥控器掉线检测 (1kHz 调用)
 * @return 无
 *
 * flag 由接收中断置 1，本函数由状态任务按 1 kHz 调用。
 * 连续约 100 ms 未收到帧时清除 rc_online，恢复收帧后自动置 1。
 */
void RC_OnlineCheck(void)
{
    if (flag)
    {
        flag = 0;               // 清狗, 等待下一帧再喂
        rc_online = 1;          // 有帧 → 在线
        error_count = 0;
    }
    else
    {
        if (error_count < 100U) error_count++;   // 状态任务每毫秒计一次，封顶防溢出
        if (error_count >= 100U) rc_online = 0;  // 约 100 ms 无帧则判掉线
    }
}
